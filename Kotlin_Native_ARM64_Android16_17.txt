    compileSdk = 37

    defaultConfig {
        applicationId = "com.example.arm64native"
        minSdk = 36
        targetSdk = 37

        // Package native libraries for ARM64 devices only.
        ndk {
            abiFilters += "arm64-v8a"
        }

        versionCode = 1
        versionName = "1.0"
    }

    // Keep the project limited to the requested ARM64 ABI.
    splits {
        abi {
            isEnable = true
            reset()
            include("arm64-v8a")
            isUniversalApk = false

    compileOptions {
        sourceCompatibility = JavaVersion.VERSION_17
        targetCompatibility = JavaVersion.VERSION_17
    }

    kotlinOptions {
        jvmTarget = "17"
    }

            isMinifyEnabled = false
        }
    }

---------------------------------------------------------------
package com.example.arm64native

import android.app.Activity
import android.os.Build
import android.os.Bundle

/**
 * Minimal native Kotlin entry point.
 *
 * Supported Android API range: 36 through 37.
 * The Gradle configuration packages ARM64 (arm64-v8a) only.
 */
class MainActivity : Activity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        // Enforce the requested Android 16–17 runtime range.
        if (Build.VERSION.SDK_INT < 36 || Build.VERSION.SDK_INT > 37) {
            finish()
            return
        }

        // No additional application feature or UI is implemented.
    }
}

IMPORTANT BUILD NOTES
---------------------
1. Install Android SDK Platform 37 and the Android SDK Build-Tools required by
   the selected Android Gradle Plugin in Android Studio.
2. Use JDK 17 for this project configuration.
3. The ABI filter and ABI split restrict packaged native ABI output to
   arm64-v8a. This does not itself restrict the physical device form factor.
4. AndroidManifest.xml declares maxSdkVersion 37 as an installation limit.
   Android tooling may warn that maxSdkVersion is generally discouraged;
   the Kotlin runtime check also exits on API levels outside 36–37.
5. Mobile and tablet support is left to Android's normal compatibility
   behavior. No TV, Wear OS, or other device-specific feature is declared.
6. This text file contains the project source/configuration files. It is not
   itself an APK and does not include the Gradle wrapper JAR or SDK binaries.
