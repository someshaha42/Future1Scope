KOTLIN-ONLY NATIVE ANDROID APPLICATION
========================================
Standalone project blueprint: Kotlin + Jetpack Compose
Fixed target: ARM64 (arm64-v8a), Mobile + Tablet, Android API 36–37

IMPORTANT SCOPE
---------------
- Application screens, UI components, navigation, state, settings, and app logic
  are written in Kotlin.
- Jetpack Compose is used instead of XML layout files.
- AndroidManifest.xml and Gradle Kotlin DSL files remain necessary project
  configuration files; they are not application UI/business logic.
- Android SDK APIs are used for operating-system features.
- This is a clean starter application architecture, not a finished screen
  recorder, editor, or other feature-rich product. Feature modules below are
  extension points and do not pretend to implement features that were not
  specified.
- The project has no connection to any previous project.

FIXED CONFIGURATION
-------------------
Programming language: Kotlin
UI framework: Jetpack Compose (Material 3)
Native platform: Android
ABI: arm64-v8a only
Device categories: Mobile + Tablet
Minimum SDK: 36 (Android 16)
Maximum supported SDK: 37 (Android 17)
Target SDK: 37
Java/JVM target: 17

package com.example.kotnative

import android.os.Build
import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import com.example.kotnative.app.AppRoot
import com.example.kotnative.ui.theme.KotlinNativeTheme

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        // Runtime guard complements the manifest installation compatibility.
        if (Build.VERSION.SDK_INT !in 36..37) {
            finish()
            return
        }

        setContent {
            KotlinNativeTheme {
                AppRoot()
            }
        }
    }
}

package com.example.kotnative.app

import androidx.lifecycle.ViewModel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow

data class AppUiState(
    val title: String = "Kotlin Native Application",
    val description: String =
        "Standalone Kotlin + Jetpack Compose project is ready."
)

class AppViewModel : ViewModel() {
    private val _state = MutableStateFlow(AppUiState())
    val state: StateFlow<AppUiState> = _state.asStateFlow()

    fun refresh() {
        _state.value = _state.value.copy(
            description = "Application state refreshed."
        )
    }
}

package com.example.kotnative.ui

import androidx.compose.foundation.layout.Arrangement
import androidx.compose.foundation.layout.Column
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.foundation.layout.padding
import androidx.compose.material3.Button
import androidx.compose.material3.Scaffold
import androidx.compose.material3.Text
import androidx.compose.material3.TopAppBar
import androidx.compose.material3.ExperimentalMaterial3Api
import androidx.compose.runtime.Composable
import androidx.compose.runtime.collectAsState
import androidx.compose.runtime.getValue
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import com.example.kotnative.app.AppUiState
import kotlinx.coroutines.flow.StateFlow

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun AppScreen(
    state: StateFlow<AppUiState>,
    onRefresh: () -> Unit
) {
    val uiState by state.collectAsState()

    Scaffold(
        topBar = {
            TopAppBar(title = { Text(uiState.title) })
        }
    ) { innerPadding ->
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(innerPadding)
                .padding(24.dp),
            verticalArrangement = Arrangement.spacedBy(16.dp)
        ) {
            Text(uiState.description)
            Button(onClick = onRefresh) {
                Text("Refresh")
            }
        }
    }
}

package com.example.kotnative.data

import android.content.Context
import androidx.datastore.preferences.core.booleanPreferencesKey
import androidx.datastore.preferences.core.edit
import androidx.datastore.preferences.preferencesDataStore
import kotlinx.coroutines.flow.map

private val Context.settingsDataStore by preferencesDataStore(
    name = "app_settings"
)

class SettingsRepository(private val context: Context) {
    private val notificationsKey =
        booleanPreferencesKey("notifications_enabled")

    val settings = context.settingsDataStore.data.map { preferences ->
        AppSettings(
            notificationsEnabled = preferences[notificationsKey] ?: true
        )
    }

    suspend fun setNotificationsEnabled(enabled: Boolean) {
        context.settingsDataStore.edit { preferences ->
            preferences[notificationsKey] = enabled
        }
    }
}

package com.example.kotnative.data

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.launch

class SettingsViewModel(
    private val repository: SettingsRepository
) : ViewModel() {
    val settings = repository.settings.stateIn(
        scope = viewModelScope,
        started = SharingStarted.WhileSubscribed(5_000),
        initialValue = AppSettings()
    )

    fun setNotificationsEnabled(enabled: Boolean) {
        viewModelScope.launch {
            repository.setNotificationsEnabled(enabled)
        }
    }
}


IMPLEMENTATION NOTES
-------------------
1. Android Studio must have Android SDK Platform 37 installed. The Android
   Gradle Plugin version must support compileSdk 37. If the selected AGP does
   not support API 37, update AGP and the matching Gradle wrapper to compatible
   versions before building.
2. The Compose BOM and AndroidX library versions above are pinned examples.
   Use a mutually compatible set if Android Studio reports a dependency or
   repository resolution error.
3. The Gradle wrapper files (gradlew, gradlew.bat, gradle/wrapper/*) are
   generated by Android Studio / Gradle and are not included in this text.
4. maxSdkVersion is generally discouraged by Android because it can block
   installation on later Android releases. It is included here because the
   requested fixed range explicitly ends at API 37. The runtime guard is a
   second check, not a replacement for installation compatibility.
5. ABI filtering controls packaged native libraries. Kotlin bytecode itself is
   not ARM64-only; the ABI restriction applies to native libraries/APK splits.
6. Android does not have a universal "tablet-only/mobile-only" manifest switch.
   This project supports phones and tablets using responsive Compose UI and
   does not declare TV, Wear OS, or Automotive features.
7. UI and app logic in this starter are Kotlin. Manifest, Gradle files and
   strings resource remain platform project files. No XML layout is used.
8. This is a compilable starting structure once compatible toolchain versions
   are selected. It is not a complete implementation of unspecified app
   features such as recording, editing, accounts, cloud sync, or export.
